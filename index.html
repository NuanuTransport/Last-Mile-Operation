/* ===================================================================== PERINGATAN ORDER TELAT KE TELEGRAM (Last Mile) File terpisah di project yang sama dengan WebAPI.gs, WaterReport.gs, LastMileReport.gs. Memakai bot & grup yang sama (WR_BOT_TOKEN, WR_CHAT_ID dari WaterReport.gs). Cara kerja: - Dicek otomatis setiap 5 menit. - Delivery yang statusnya masih "On Delivery" lebih dari 30 menit sejak Pickup Time langsung dikirim peringatan ke grup. - Kalau masih belum selesai, diingatkan lagi di menit ke-60 dan ke-90, lalu berhenti (supaya order yang lupa ditutup tidak mengirim pesan terus-menerus). - Order yang sudah lebih dari 3 jam tidak diperingatkan lagi. Setup: 1. Jalankan laPreview untuk melihat order yang sedang telat (lihat Execution log). 2. Jalankan laSetupTrigger sekali untuk mengaktifkan pengecekan setiap 5 menit. ===================================================================== */ const LA_EVERY_MIN = 5; // interval pengecekan (menit): 1, 5, 10, 15, atau 30 const LA_REMIND_MIN = 30; // jarak pengingat berikutnya (menit) const LA_MAX_ALERTS = 3; // maksimal peringatan per order (30, 60, 90 menit) const LA_IGNORE_MIN = 180; // order yang sudah lebih dari 3 jam diabaikan (kemungkinan lupa ditutup) const LA_PROP = 'LA_SENT'; /* ---------- jadwal ---------- */ function laRun() { const late = laCheck_(); const props = PropertiesService.getScriptProperties(); props.setProperty(LA_PROP, JSON.stringify(late.keep)); if (late.alerts.length) wrSend_(laMsg_(late.alerts)); } function laSetupTrigger() { ScriptApp.getProjectTriggers() .filter(t => t.getHandlerFunction() === 'laRun') .forEach(t => ScriptApp.deleteTrigger(t)); ScriptApp.newTrigger('laRun').timeBased().everyMinutes(LA_EVERY_MIN).create(); Logger.log('Trigger aktif: laRun setiap ' + LA_EVERY_MIN + ' menit'); } /* ---------- uji coba (tidak mengirim, tidak menyimpan) ---------- */ function laPreview() { const late = laCheck_(true); Logger.log(late.alerts.length ? laMsg_(late.alerts) : 'Tidak ada order telat saat ini.'); } /* ---------- inti ---------- */ function laCheck_(all) { const ss = SpreadsheetApp.getActive(); const tz = ss.getSpreadsheetTimeZone(); const now = new Date(); const sent = JSON.parse(PropertiesService.getScriptProperties().getProperty(LA_PROP) || '{}'); const keep = {}, alerts = []; readSheet_(ss.getSheetByName(SHEET_NAME), tz) .filter(o => o['System ID'] && String(o['Status'] || '').trim() === DELIVERY_ACTIVE) .forEach(o => { const start = laStart_(o, tz); if (!start) return; const min = Math.floor((now - start) / 60000); const id = String(o['System ID']); const done = sent[id] || 0; if (done) keep[id] = done; if (min < SLA_MINUTES || min > LA_IGNORE_MIN) return; const level = Math.min(LA_MAX_ALERTS, Math.floor((min - SLA_MINUTES) / LA_REMIND_MIN) + 1); if (all || level > done) { alerts.push({ o: o, min: min, level: level, start: start }); keep[id] = Math.max(done, level); } }); return { alerts: alerts, keep: keep }; } // Waktu mulai = tanggal (kolom Date) + jam dari Pickup Time (mendukung "1:15:00 PM") function laStart_(o, tz) { const raw = String(o['Pickup Time'] || ''); const dm = raw.match(/\d{4}-\d{2}-\d{2}/); const date = dm ? dm[0] : String(o['Date'] || '').slice(0, 10); const hm = lrHM_(raw); if (!/^\d{4}-\d{2}-\d{2}$/.test(date) || !hm) return null; return Utilities.parseDate(date + ' ' + hm, tz, 'yyyy-MM-dd HH:mm'); } function laMsg_(alerts) { const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone(); const blocks = alerts.sort((a, b) => b.min - a.min).map(x => { const o = x.o; return [ wrPad_('Order', 9) + (o['Order ID'] || o['System ID']), wrPad_('Driver', 9) + (o['Driver Last Mile'] || '-'), wrPad_('To', 9) + (o['Dropoff'] || '-'), wrPad_('Vendor', 9) + (o['Vendor'] || '-'), wrPad_('Pickup', 9) + Utilities.formatDate(x.start, tz, 'HH:mm') + ' (' + x.min + ' min ago)' ].join('\n'); }); const first = alerts.every(x => x.level === 1); return 'LATE DELIVERY ALERT\n' + '' + (first ? 'Still on delivery after ' + SLA_MINUTES + ' min (SLA).' : 'Still not delivered. Please follow up.') + '\n\n' + '
' + wrEsc_(blocks.join('\n\n')) + '
\n' + 'Send the POD in the web app once delivered.'; }
